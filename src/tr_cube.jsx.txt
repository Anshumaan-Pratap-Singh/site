/* Cube Folding: pick the cube a flat net folds into, as in SHL / Flow Traders / IMC spatial reasoning tests.
   The net is folded in 3D for real, so wrong options are genuinely impossible: two opposite faces shown
   together, or three neighbouring faces in mirror-image order. */
(function () {
  const { TH } = TIP;
  const { trainer } = TIP_DRILL;
  const K = TIP_KIT;
  const NETS = [
    [[1, 0], [0, 1], [1, 1], [2, 1], [1, 2], [1, 3]], [[0, 0], [1, 0], [2, 0], [1, 1], [1, 2], [1, 3]], [[0, 0], [0, 1], [1, 1], [2, 1], [3, 1], [3, 2]],
    [[1, 0], [0, 1], [1, 1], [2, 1], [3, 1], [2, 2]], [[0, 0], [1, 0], [1, 1], [2, 1], [2, 2], [3, 2]], [[0, 0], [1, 0], [1, 1], [2, 1], [3, 1], [2, 2]],
    [[0, 1], [1, 1], [2, 1], [3, 1], [0, 0], [3, 2]], [[1, 0], [1, 1], [0, 1], [2, 1], [2, 2], [3, 2]], [[0, 0], [1, 0], [2, 0], [2, 1], [3, 1], [4, 1]],
    [[0, 1], [1, 1], [1, 0], [2, 0], [3, 0], [2, 1]], [[0, 0], [0, 1], [1, 1], [2, 1], [2, 2], [3, 1]],
  ];
  const FACES = [["circle", 1, "half"], ["square", 2, "half"], ["cross", 3, "half"], ["diamond", 4, "half"], ["star", 5, "half"], ["circle", 2, "solid"]];
  const neg = v => v.map(x => -x), eq = (a, b) => a[0] === b[0] && a[1] === b[1] && a[2] === b[2];
  /* fold: walk the net from the first cell carrying the 3D frame (normal, right, down) */
  function fold(cells) {
    const at = {}, key = (x, y) => x + "," + y;
    cells.forEach((c, i) => at[key(c[0], c[1])] = i);
    const fr = { 0: { n: [0, 0, -1], r: [1, 0, 0], d: [0, 1, 0] } }, q = [0];
    while (q.length) {
      const i = q.shift(), [x, y] = cells[i], f = fr[i];
      [[1, 0, o => ({ n: o.r, r: neg(o.n), d: o.d })], [-1, 0, o => ({ n: neg(o.r), r: o.n, d: o.d })], [0, 1, o => ({ n: o.d, r: o.r, d: neg(o.n) })], [0, -1, o => ({ n: neg(o.d), r: o.r, d: o.n })]]
        .forEach(([dx, dy, mv]) => { const j = at[key(x + dx, y + dy)]; if (j != null && fr[j] == null) { fr[j] = mv(f); q.push(j); } });
    }
    const normals = cells.map((c, i) => fr[i].n);
    const ok = normals.every((n, i) => normals.findIndex(m => eq(m, n)) === i);
    return ok ? normals : null;
  }
  const NET = cells => <K.Art w={210} h={170} label="cube net">{(() => {
    const mx = Math.max(...cells.map(c => c[0])) + 1, my = Math.max(...cells.map(c => c[1])) + 1, s = Math.min(200 / mx, 160 / my), ox = (210 - mx * s) / 2, oy = (170 - my * s) / 2;
    return cells.map((c, i) => <g key={i}><rect x={ox + c[0] * s} y={oy + c[1] * s} width={s} height={s} rx="4" className="tile" />
      <K.Shape kind={FACES[i][0]} x={ox + c[0] * s + s / 2} y={oy + c[1] * s + s / 2} r={s * 0.28} tone={FACES[i][1]} fill={FACES[i][2]} /></g>);
  })()}</K.Art>;
  function Iso({ f }) {   /* f = [top, right, left] face indices */
    const T = "M50,8L88,28L50,48L12,28z", Rr = "M50,48L88,28V72L50,92z", L = "M12,28L50,48V92L12,72z";
    const c = [[50, 28], [69, 60], [31, 60]];
    return <K.Art w={100} h={100}><path d={T} className="tile" /><path d={Rr} className="tile" style={{ filter: "brightness(.94)" }} /><path d={L} className="tile" style={{ filter: "brightness(.88)" }} />
      {f.map((fi, k) => <K.Shape key={k} kind={FACES[fi][0]} x={c[k][0]} y={c[k][1]} r={12} tone={FACES[fi][1]} fill={FACES[fi][2]} />)}</K.Art>;
  }
  function gen(r, s) {
    let cells, nm;
    for (let t = 0; t < 30 && !nm; t++) { cells = r.pick(NETS); const order = r.shuffle([0, 1, 2, 3, 4, 5]); cells = order.map(i => cells[i]); nm = fold([cells[0], ...cells.slice(1)]); }
    const faceOf = v => nm.findIndex(m => eq(m, v));
    const sx = r.pick([1, -1]), sy = r.pick([1, -1]), sz = r.pick([1, -1]);
    const A = faceOf([sx, 0, 0]), B = faceOf([0, sy, 0]), C = faceOf([0, 0, sz]);
    const det = sx * sy * sz;
    const right = det > 0 ? [A, C, B] : [A, B, C];
    const opp = i => faceOf(neg(nm[i]));
    const key = f => f.join("");
    const wrong = new Map();
    const add = f => { if (key(f) !== key(right) && new Set(f).size === 3) wrong.set(key(f), f); };
    add([right[0], right[2], right[1]]);                         /* mirror image */
    for (let t = 0; t < 30 && wrong.size < 5; t++) { const k = r.int(0, 2), f = right.slice(); f[k] = opp(f[k]); add(r() < 0.5 ? f : [f[0], f[2], f[1]]); }
    const opts = r.shuffle([right, ...r.shuffle([...wrong.values()]).slice(0, s.options - 1)]);
    return {
      text: "Which cube can this net fold into?", answer: "the option shown in green",
      explain: "Faces opposite each other on the net (with one face between them in a line) can never be seen together, and three faces around a corner keep their clockwise order.",
      prompt: <div className="col" style={{ alignItems: "center", gap: 6 }}><div className="oa-lbl">Which cube can this net fold into?</div>{NET(cells)}</div>,
      input: "choice", cols: Math.min(4, opts.length), choiceCls: "cb-opts", choices: opts.map(f => ({ value: key(f), label: key(f), node: <Iso f={f} /> })), check: v => v === key(right),
    };
  }
  TIP_TRAINERS.register(trainer({
    id: "cube", name: "Cube Folding", version: 2,
    blurb: "The spatial test from SHL, Flow Traders and IMC assessments: a flat net is folded in 3D for real, and you pick the only possible cube. Wrong options show opposite faces together or three faces in mirror-image order.",
    fields: [
      { key: "count", label: "Questions", type: "num", min: 5, max: 30 },
      { key: "time", label: "Time per question", type: "num", min: 6, max: 90, unit: "s" },
      { key: "options", label: "Options", type: "num", min: 3, max: 5 },
    ],
    presets: { Easy: { count: 8, time: 45, options: 3 }, Medium: { count: 12, time: 25, options: 4 }, Hard: { count: 16, time: 15, options: 5 } },
    note: "First rule out options with two opposite faces; then check the clockwise order around the corner.",
  }, { gen, count: s => s.count, perQSecs: s => s.time, always: true }));
})();
