def once(text, old, new, what):
    base = text.find('/* ---- end TIP_G2 ---- */')
    head, tail = (text[:base], text[base:]) if base >= 0 and old != 'var TIP_MM = (function() {' else ('', text)
    n = tail.count(old)
    if n != 1:
        sys.exit(f'{what}: expected 1 match, found {n}')
    return head + tail.replace(old, new)


def span(text, start, end, new, what, keep_end=True):
    """Replace text from `start` up to (not including, if keep_end) `end`; both must be unique after start."""
    base = text.find('/* ---- end TIP_G2 ---- */') if not start.startswith('/* ---- ') else 0
    i = text.find(start, max(0, base))
    if i < 0 or text.find(start, i + 1) >= 0:
        sys.exit(f'{what}: start marker not unique')
    j = text.find(end, i)
    if j < 0:
        sys.exit(f'{what}: end marker not found')
    return text[:i] + new + (text[j:] if keep_end else text[j + len(end):])
