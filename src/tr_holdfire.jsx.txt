/* Hold Fire: go / no-go with rule changes, as in the response-inhibition games of Pymetrics-style assessments.
   Shapes flash one at a time; tap (Space) only for the targets in the current rule and hold back for the rest.
   The rule changes during the session and the window shrinks. Scored on hits, false alarms and d′. */
(function () {
  const { TH, TF, R, typing } = TIP;
  const { useState, useEffect, useMemo, useRef } = R;
  const K = TIP_KIT;
  const SH = ["circle", "square", "triangle", "star"], COL = [[3, "green"], [4, "red"], [1, "blue"], [2, "amber"]];
  const RULES = [
    { t: "Tap for GREEN shapes", go: x => x.c === 3 }, { t: "Tap for CIRCLES", go: x => x.k === "circle" },
    { t: "Tap for everything except RED", go: x => x.c !== 4 }, { t: "Tap for BLUE or STARS", go: x => x.c === 1 || x.k === "star" },
    { t: "Tap for everything except SQUARES", go: x => x.k !== "square" }, { t: "Tap for AMBER TRIANGLES", go: x => x.c === 2 && x.k === "triangle" },
  ];
  const z = p => { p = Math.min(0.99, Math.max(0.01, p)); const a = [2.50662823884, -18.61500062529, 41.39119773534, -25.44106049637], b = [-8.4735109309, 23.08336743743, -21.06224101826, 3.13082909833];
    const q = p - 0.5; if (Math.abs(q) < 0.42) { const r = q * q; return q * (((a[3] * r + a[2]) * r + a[1]) * r + a[0]) / ((((b[3] * r + b[2]) * r + b[1]) * r + b[0]) * r + 1); } const r = Math.sqrt(-Math.log(q < 0 ? p : 1 - p)), v = 0.3374754822726147 + r * (0.9761690190917186 + r * (0.1607979714918209 + r * 0.0276438810333863)); return q < 0 ? -v : v; };
  function Game({ settings: s, onFinish, onRestart }) {
    const [rng] = useState(() => jt());
    const [order] = useState(() => rng.shuffle(RULES.slice()));
    const [i, setI] = useState(-1), [stim, setStim] = useState(null), [rule, setRule] = useState(0), [banner, setBanner] = useState(true);
    const [log, setLog] = useState([]), [flash, setFlash] = useState(null), [done, setDone] = useState(false);
    const resp = useRef(false), cur = useRef(null);
    const W0 = s.window * 1000, W1 = s.minWin * 1000;
    const win = k => Math.max(W1, W0 - (W0 - W1) * k / s.count);
    useEffect(() => { setI(0); }, []);
    useEffect(() => { if (!banner) return; const t = setTimeout(() => setBanner(false), 1700); return () => clearTimeout(t); }, [banner]);
    useEffect(() => {
      if (i < 0 || done || banner) return;
      if (i >= s.count) { setDone(true); return; }
      const ri = Math.floor(i / s.every) % order.length;
      if (ri !== rule) { setRule(ri); setBanner(true); return; }
      /* about 65% targets */
      let x; const want = rng() < s.goP;
      for (let t = 0; t < 40; t++) { x = { k: rng.pick(SH), c: rng.pick(COL)[0] }; if (order[ri].go(x) === want) break; }
      x.go = order[ri].go(x); cur.current = x; resp.current = false; setStim(x);
      const w = win(i);
      const t1 = setTimeout(() => {
        const r = resp.current, ok = x.go === r;
        setLog(l => [...l, { go: x.go, r, ok }]);
        if (!r) setFlash(x.go ? "miss" : null);
        setStim(null); cur.current = null;
        setTimeout(() => setFlash(null), 250);
      }, w);
      const t2 = setTimeout(() => setI(n => n + 1), w + s.gap);
      return () => { clearTimeout(t1); clearTimeout(t2); };
    }, [i, banner]);
    const press = () => { if (!cur.current || resp.current || banner) return; resp.current = true; setFlash(cur.current.go ? "hit" : "fa"); };
    useEffect(() => { const h = e => { if (e.key === " " && !typing(e)) { e.preventDefault(); press(); } }; window.addEventListener("keydown", h); return () => window.removeEventListener("keydown", h); }, [banner]);
    const res = useMemo(() => {
      if (!done) return null;
      const H = log.filter(l => l.go && l.r).length, M = log.filter(l => l.go && !l.r).length, FA = log.filter(l => !l.go && l.r).length, CR = log.filter(l => !l.go && !l.r).length;
      const hr = (H + 0.5) / (H + M + 1), far = (FA + 0.5) / (FA + CR + 1);
      return { score: H + CR - 2 * FA, hits: H, misses: M, fa: FA, cr: CR, dprime: Math.round((z(hr) - z(far)) * 100) / 100, correct: H + CR, total: log.length };
    }, [done]);
    Jt(done, res, onFinish);
    if (done && res) return <K.Result headline={`${res.score} points · d′ ${res.dprime}`} sub={`${res.hits} hits, ${res.fa} false alarms, ${res.misses} misses`} ring={res.total ? res.correct / res.total : 0} onRestart={onRestart}
      metrics={[{ label: "Hits", value: res.hits, tone: "up" }, { label: "Correct holds", value: res.cr }, { label: "False alarms", value: res.fa, tone: res.fa ? "dn" : "" }, { label: "Misses", value: res.misses }, { label: "d′", value: res.dprime, sub: "sensitivity; 2+ is strong" }]}
      notes="Score = hits + correct holds − 2 × false alarms. The rule changes every few shapes; read it fresh each time instead of carrying the old one." />;
    const ok = log.filter(l => l.ok).length;
    return (
      <K.Stage title="Hold Fire" sub={`Shape ${Math.max(0, Math.min(i + 1, s.count))} of ${s.count}`} stats={[{ label: "Correct", value: ok, pop: true }, { label: "False alarms", value: log.filter(l => !l.go && l.r).length, tone: "dn" }]} total={Math.min(s.count, 80)} results={log.map(l => l.ok)} at={i}>
        <div className="hf-wrap" onPointerDown={e => { e.preventDefault(); press(); }}>
          <div className={"hf-rule" + (banner ? " big" : "")}>{order[rule].t}</div>
          <div className={"hf-stage" + (flash ? " " + flash : "")}>
            {stim && !banner ? <K.Art w={140} h={140} key={i}><K.Shape kind={stim.k} x={70} y={70} r={48} fill="solid" tone={stim.c} /></K.Art> : <div className="hf-dot" />}
          </div>
          <div className="small muted" style={{ textAlign: "center" }}>Press <kbd>Space</kbd> or tap here for a target. Do nothing for anything else.</div>
        </div>
      </K.Stage>
    );
  }
  TIP_TRAINERS.register({
    id: "holdfire", name: "Hold Fire", metric: "score", unit: " pts", version: 2, playFilter: K.versioned(2),
    blurb: "Go / no-go with changing rules, like the response-inhibition games in Pymetrics-style assessments: shapes flash one at a time; tap only for the current targets and hold back for everything else while the window shrinks. Scored on hits, false alarms and d′.",
    fields: [
      { key: "count", label: "Shapes", type: "num", min: 20, max: 120 },
      { key: "window", label: "Starting window", type: "num", min: 0.5, max: 2.5, unit: "s", step: 0.1 },
      { key: "minWin", label: "Final window", type: "num", min: 0.3, max: 1.5, unit: "s", step: 0.1 },
      { key: "every", label: "Rule changes every", type: "num", min: 8, max: 60, unit: " shapes" },
    ],
    presets: { Easy: { count: 30, window: 1.4, minWin: 1.0, every: 30, gap: 450, goP: 0.65 }, Medium: { count: 48, window: 1.1, minWin: 0.7, every: 16, gap: 350, goP: 0.65 }, Hard: { count: 72, window: 0.9, minWin: 0.5, every: 12, gap: 300, goP: 0.7 } },
    fix: c => { if (c.gap == null) c.gap = 350; if (c.goP == null) c.goP = 0.65; },
    note: "Targets are about two-thirds of the shapes, so the urge to tap is strong: that is the point.",
    component: Game,
  });
})();
