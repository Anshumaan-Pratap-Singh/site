"""Build app.js and index.html for Trading Interview Practice.

One script does the whole build (since 19 Nov 2026 it also applies the dashboard
and learning patches that used to be patched straight into the live app.js):

  python3 src/build.py [--esbuild PATH] [--out DIR]

Inputs, all relative to the artifact root (the folder above src/):
  src/base-app.js, src/base-index.html   original Tradermath bundle and page
  src/*.jsx(.txt), src/learn_data.js(.txt) new modules (".txt" copies are what the artifact stores)
  src/tip_dash.js(.txt)                   Dashboard component, replaces `function em`
Outputs: app.js and index.html in --out (default: the artifact root).
Every patch must match exactly once or the build stops, so a changed base is caught.
"""
import argparse, pathlib, subprocess, sys

OUT = pathlib.Path(args.out)
OUT.mkdir(parents=True, exist_ok=True)


def src(name):
    """Read src/<name>, falling back to src/<name>.txt (the artifact's stored copy)."""
    for p in (SRC / name, SRC / (name + '.txt')):
        if p.exists():
            return p
    sys.exit(f'missing source file: src/{name}')


def jsx_names():
    names = {p.name[:-4] if p.name.endswith('.txt') else p.name for p in SRC.iterdir()}
    return sorted(n for n in names if n.startswith('tr_') and n.endswith('.jsx'))


def compile_js(name):
    r = subprocess.run([args.esbuild, '--loader=jsx', '--jsx-factory=TH', '--jsx-fragment=TF',
                        '--target=es2020', '--charset=utf8', '--log-level=warning'],
                       input=src(name).read_text(encoding='utf8'), capture_output=True, text=True)
    if r.returncode:
        sys.exit(f'{name}: {r.stderr}')
    if r.stderr.strip():
        print(r.stderr)
    return r.stdout


class Patcher:
    def __init__(self, s):
        self.s = s

    def __call__(self, old, new, count=1):
        n = self.s.count(old)
        if n != count:
            sys.exit(f'patch expected {count} match(es), found {n}: {old[:90]!r}')
        self.s = self.s.replace(old, new)
