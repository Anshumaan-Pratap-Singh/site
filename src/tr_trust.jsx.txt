/* Trust Game: a repeated pricing dilemma like the trust and cooperation games in Pymetrics-style assessments.
   You and a rival seller each price High or Low every round. Both High: 3 each. You Low, rival High: you 5, rival 0.
   Both Low: 1 each. The rival follows a hidden strategy; read it and maximise your total. */
(function () {
  const { TH, TF, R, typing } = TIP;
  const { useState, useEffect, useMemo } = R;
  const K = TIP_KIT;
  const PAY = { HH: [3, 3], HL: [0, 5], LH: [5, 0], LL: [1, 1] };
  const STRATS = {
    tft: { name: "Tit for tat", move: h => h.length ? h[h.length - 1].you : "H", tip: "copies your last move" },
    grudge: { name: "Grudger", move: h => h.some(x => x.you === "L") ? "L" : "H", tip: "cooperates until you undercut once, then never again" },
    gen: { name: "Generous tit for tat", move: (h, r) => !h.length ? "H" : h[h.length - 1].you === "L" && r() > 0.33 ? "L" : "H", tip: "copies you but forgives a third of undercuts" },
    rand: { name: "Random", move: (h, r) => r() < 0.5 ? "H" : "L", tip: "prices at random, so undercutting is best" },
    shark: { name: "Exploiter", move: (h, r) => h.length % 4 === 3 ? "L" : h.length > 1 && h[h.length - 1].you === "L" ? "L" : "H", tip: "cooperates but undercuts every fourth round" },
  };
  function Game({ settings: s, onFinish, onRestart }) {
    const [rng] = useState(() => jt());
    const [seq] = useState(() => Array.from({ length: s.rivals }, () => rng.pick(Object.keys(STRATS))));
    const [ri, setRi] = useState(0), [h, setH] = useState([]), [all, setAll] = useState([]), [fb, setFb] = useState(null), [done, setDone] = useState(false);
    const strat = STRATS[seq[ri]];
    function play(you) {
      if (fb || done) return;
      const rival = strat.move(h, rng), [a, b] = PAY[you + rival];
      const nh = [...h, { you, rival, a, b }];
      setH(nh); setFb({ you, rival, a, b });
      setTimeout(() => {
        setFb(null);
        if (nh.length >= s.rounds) {
          const na = [...all, { strat: seq[ri], h: nh }]; setAll(na);
          if (ri + 1 >= s.rivals) { setDone(true); return; }
          setRi(ri + 1); setH([]);
        }
      }, 800);
    }
    useEffect(() => { const k = e => { if (typing(e)) return; if (e.key === "h") play("H"); if (e.key === "l") play("L"); }; window.addEventListener("keydown", k); return () => window.removeEventListener("keydown", k); });
    const res = useMemo(() => {
      if (!done) return null;
      const you = all.reduce((t, r) => t + r.h.reduce((a, x) => a + x.a, 0), 0), max = all.length * s.rounds * 3;
      return { score: you, max, rivals: all.map(r => ({ strat: r.strat, you: r.h.reduce((a, x) => a + x.a, 0), them: r.h.reduce((a, x) => a + x.b, 0) })) };
    }, [done]);
    Jt(done, res, onFinish);
    if (done && res) return <K.Result headline={`${res.score} points`} sub={`Mutual cooperation every round would give ${res.max}`} ring={Math.min(1, res.score / res.max)} onRestart={onRestart}
      metrics={res.rivals.map((r, i) => ({ label: `Rival ${i + 1}: ${STRATS[r.strat].name}`, value: `${r.you} – ${r.them}`, sub: STRATS[r.strat].tip }))}
      notes="Against copiers, cooperate and they cooperate back; undercutting wins one round and costs the rest. Against a random or exploiting rival, protect yourself. Spotting which rival you face is the skill." />;
    const you = h.reduce((a, x) => a + x.a, 0), them = h.reduce((a, x) => a + x.b, 0);
    return (
      <K.Stage title="Trust Game" sub={`Rival ${ri + 1} of ${s.rivals} · round ${Math.min(h.length + 1, s.rounds)} of ${s.rounds}`} stats={[{ label: "You", value: you, pop: true, tone: "up" }, { label: "Rival", value: them }]} total={s.rounds} results={h.map(x => x.a >= x.b)} at={h.length}>
        <div className="oa-q">
          <div className="tr-pay"><div className="oa-lbl">Payoffs (you, rival)</div>
            <table><thead><tr><th></th><th>Rival High</th><th>Rival Low</th></tr></thead><tbody><tr><th>You High</th><td>3, 3</td><td>0, 5</td></tr><tr><th>You Low</th><td>5, 0</td><td>1, 1</td></tr></tbody></table></div>
          <div className="tr-hist">{h.map((x, i) => <div key={i} className="tr-col"><b className={x.you === "H" ? "h" : "l"}>{x.you}</b><b className={x.rival === "H" ? "h" : "l"}>{x.rival}</b></div>)}
            {fb ? null : <div className="tr-col q"><b>?</b><b>?</b></div>}</div>
          {fb ? <K.Banner ok={fb.a >= fb.b} title={`You ${fb.you === "H" ? "priced High" : "undercut"}, rival ${fb.rival === "H" ? "priced High" : "undercut"}.`}>You +{fb.a}, rival +{fb.b}.</K.Banner>
            : <K.Choices items={[{ value: "H", node: <span>Price High (cooperate)</span> }, { value: "L", node: <span>Price Low (undercut)</span> }]} cols={2} onPick={play} />}
        </div>
      </K.Stage>
    );
  }
  TIP_TRAINERS.register({
    id: "trust", name: "Trust Game", metric: "score", unit: " pts", version: 2, playFilter: K.versioned(2),
    blurb: "A repeated pricing dilemma like the trust and cooperation games in Pymetrics-style assessments: you and a rival price High or Low each round. Cooperation pays steadily, undercutting pays once, and each rival follows a hidden strategy you have to read.",
    fields: [{ key: "rivals", label: "Rivals", type: "num", min: 1, max: 5 }, { key: "rounds", label: "Rounds per rival", type: "num", min: 5, max: 20 }],
    presets: { Easy: { rivals: 2, rounds: 8 }, Medium: { rivals: 3, rounds: 10 }, Hard: { rivals: 4, rounds: 12 } },
    note: "Keys: H or 1 = High, L or 2 = Low. You don't know how many rounds the rival expects, so there is no safe last-round grab.",
    component: Game,
  });
})();
