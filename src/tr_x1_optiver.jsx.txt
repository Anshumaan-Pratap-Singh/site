  /* five options, the answer is the closest one: spaced at least 20% apart so "closest" is never ambiguous */
  function closest(r, a, isP) {
    const fs = r.shuffle([0.45, 0.6, 0.75, 1.3, 1.55, 1.9, 2.4]);
    const out = [a];
    for (const f of fs) { let v = a * f; if (isP && v >= 1) continue; if (out.every(x => Math.abs(Math.log(Math.abs(v) / Math.abs(x || 1e-9))) > 0.18)) out.push(v); if (out.length === 5) break; }
    while (out.length < 5) out.push(a + out.length * (Math.abs(a) || 1) * 0.5);
    return r.shuffle(out).map(v => ({ value: String(v), label: isP ? pct(v) : sig(v) }));
  }

  /* ============ Intervals: a lower and upper bound; tight and right scores best ============ */
  const ipts = (lo, hi, a, isP) => { if (!(lo <= a + 1e-12 && a <= hi + 1e-12)) return 0; const scale = isP ? 1 : Math.max(Math.abs(a), 1); return Math.max(1, Math.min(10, Math.round(10 * (1 - (hi - lo) / scale)))); };

  function Range({ submit, running, isP }) {
    const [lo, setLo] = TIP.R.useState(""), [hi, setHi] = TIP.R.useState(""), [act, setAct] = TIP.R.useState("lo"), [touch] = TIP.R.useState(K.isTouch);

  /* ============ Orderbooks: read a book, pick the profitable trade ============ */
  function book(r) {
    const mid = r.int(40, 160) + (r() < 0.5 ? 0.5 : 0), tick = 0.5, gap = r.int(1, 2) * tick;
    const asks = [], bids = [];
    let a = mid + gap / 2 + (r() < 0.5 ? 0 : tick / 2), b = a - gap;
    a = Math.round(a * 2) / 2; b = Math.round(b * 2) / 2; if (b >= a) b = a - tick;
    for (let k = 0; k < 4; k++) { asks.push({ p: a + k * tick * r.int(1, 2), q: r.int(1, 8) * 5 }); bids.push({ p: b - k * tick * r.int(1, 2), q: r.int(1, 8) * 5 }); }
    for (let k = 1; k < 4; k++) { if (asks[k].p <= asks[k - 1].p) asks[k].p = asks[k - 1].p + tick; if (bids[k].p >= bids[k - 1].p) bids[k].p = bids[k - 1].p - tick; }
    return { asks, bids };
  }
  const walkBook = (levels, qty) => { let left = qty, cost = 0; for (const l of levels) { const t = Math.min(left, l.q); cost += t * l.p; left -= t; if (!left) break; } return left ? null : cost; };
  function Book({ b }) {
    return (
      <table className="tbl ob-tbl"><thead><tr><th>Bid size</th><th>Bid</th><th>Ask</th><th>Ask size</th></tr></thead>
        <tbody>{b.asks.map((_, k) => <tr key={k}><td>{b.bids[k].q}</td><td className="pos">{b.bids[k].p.toFixed(1)}</td><td className="neg">{b.asks[k].p.toFixed(1)}</td><td>{b.asks[k].q}</td></tr>)}</tbody></table>
    );
  }
  function genOb(r) {
    const b = book(r), kind = r.pick(["best", "cost", "proceeds", "count", "unwind", "mid"]);
    let q, a, why, ch;
    if (kind === "best") {
      const F = Math.round(((b.asks[0].p + b.bids[0].p) / 2 + (r() < 0.5 ? -1 : 1) * r.int(3, 6) * 0.5) * 2) / 2;
      const trades = [...b.asks.slice(0, 3).map(l => ({ t: `Buy ${l.q} at ${l.p.toFixed(1)}`, v: (F - l.p) * l.q })), ...b.bids.slice(0, 3).map(l => ({ t: `Sell ${l.q} at ${l.p.toFixed(1)}`, v: (l.p - F) * l.q }))];
      const best = trades.reduce((x, y) => y.v > x.v ? y : x);
      if (trades.filter(t => Math.abs(t.v - best.v) < 1e-9).length > 1) return genOb(r);
      const pickd = r.shuffle(trades.filter(t => t !== best)).slice(0, 3);
      ch = r.shuffle([best, ...pickd]).map(t => ({ value: t.t, label: t.t }));
      q = `Fair value is ${F.toFixed(1)}. Which single trade against this book makes the most money?`; a = best.t; why = `Profit = size × distance from fair: ${trades.map(t => t.t + " → " + sig(t.v)).join(", ")}.`;
      return { b, q, a, why, ch, check: v => v === a, kind };
    }
    if (kind === "cost" || kind === "proceeds") {
      const side = kind === "cost" ? b.asks : b.bids, tot = side.reduce((x, l) => x + l.q, 0), qty = Math.min(tot, (r.int(2, 6) * 5) + side[0].q);
      const v = walkBook(side, qty); a = v;
      q = kind === "cost" ? `You buy ${qty} lots at market. What is the total cost?` : `You sell ${qty} lots at market. What are the total proceeds?`;
      why = `Walk the ${kind === "cost" ? "asks" : "bids"} from the best price until ${qty} lots are filled.`;
    } else if (kind === "count") {
      const F = b.asks[r.int(1, 3)].p + 0.25; a = b.asks.filter(l => l.p < F).reduce((x, l) => x + l.q, 0);
      q = `Fair value is ${F.toFixed(2)}. How many lots can you buy from this book at a profit?`; why = `Add the sizes of every ask below ${F.toFixed(2)}.`;
    } else if (kind === "unwind") {
      const pos = b.bids[0].q + b.bids[1].q; const F = Math.round(((b.asks[0].p + b.bids[0].p) / 2) * 2) / 2;
      a = walkBook(b.bids, pos) - pos * F;
      q = `You are long ${pos} lots with fair value ${F.toFixed(1)}. You must sell them all to the bids now. What is your P&L against fair value?`; why = `Proceeds from the top two bid levels minus ${pos} × ${F.toFixed(1)}.`;
    } else { a = (b.asks[0].p + b.bids[0].p) / 2; q = `What is the mid price of this book?`; why = `(best bid + best ask) / 2.`; }
    /* near misses on the book's own grid: half-ticks for prices, a few lots' worth for totals */
    const step = kind === "mid" ? 0.5 : kind === "count" ? 5 : Math.max(2.5, Math.round(Math.abs(a) * 0.004 / 2.5) * 2.5);
    const lab = v => kind === "mid" ? v.toFixed(2) : kind === "unwind" ? v.toFixed(1) : sig(v);
    ch = options(r, a, 4, (rr, x) => x + (rr() < 0.5 ? -1 : 1) * step * rr.int(1, 4)).map(v => ({ value: String(v), label: lab(v) }));
    const A = a;
    return { b, q, a: lab(A), why, ch, check: v => Math.abs(Number(v) - A) < 1e-9, kind };
  }
