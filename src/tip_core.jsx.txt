/* Trading Interview Practice: shared trainer screen, Trainers tab and the reworked Trading Game.
   Injected into app.js; uses the bundle's own helpers (Lt, se, yt, Zt, ui, jt, xl, Be, ie, Xe, Jt). */
var TIP = (function () {
  const R = oe(vt());
  const TH = R.createElement, TF = R.Fragment;
  const { useState, useEffect, useMemo, useRef } = R;
  const still = () => !!(window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches);
  const fmtDate = ts => new Date(ts).toLocaleDateString("en-GB", { day: "numeric", month: "short" });
  const fmtWhen = ts => new Date(ts).toLocaleString("en-GB", { day: "numeric", month: "short", hour: "2-digit", minute: "2-digit" });
  const DIFFS = ["Easy", "Medium", "Hard", "Custom"];
  const typing = e => { const t = e.target && e.target.tagName; return t === "INPUT" || t === "TEXTAREA" || t === "SELECT" || t === "BUTTON"; };

  /* ---------- small pieces ---------- */
  function fieldText(f, v) {
    if (f.type === "bool") return v ? "On" : "Off";
    if (f.type === "sel") { const c = f.choices.find(c => c.value === v); return c ? c.label : String(v); }
    if (f.type === "multi") { const on = f.choices.filter(c => (v || []).includes(c.value)); return on.length === f.choices.length ? "All" : on.length ? on.map(c => c.short || c.label).join(", ") : "None"; }
    return String(v) + (f.unit === " pts" && Number(v) === 1 ? " pt" : (f.unit || ""));
  }
  function clampCfg(game, cfg, diff) {
    const out = { ...cfg };
    for (const f of game.fields) {
      if (f.type !== "num") continue;
      let v = Number(out[f.key]);
      if (!isFinite(v)) v = game.presets[diff === "Custom" ? "Medium" : diff][f.key];
      v = Math.max(f.min, Math.min(f.max, v));
      if ((f.step || 1) >= 1) v = Math.round(v);
      out[f.key] = v;
    }
    if (game.fix) game.fix(out);
    return out;
  }

  function Tile({ f, value, edit, onChange }) {
    if (!edit) return <div className="tip-tile"><span>{f.label}</span><b>{fieldText(f, value)}</b></div>;
    let ctl;
    if (f.type === "bool") ctl = <button type="button" className={"tip-toggle" + (value ? " on" : "")} aria-pressed={!!value} onClick={() => onChange(!value)}><i></i>{value ? "On" : "Off"}</button>;
    else if (f.type === "multi") ctl = <span className="tip-mchips" role="group" aria-label={f.label}>{f.choices.map(c => { const on = (value || []).includes(c.value);
      return <button key={c.value} type="button" className={on ? "on" : ""} aria-pressed={on} onClick={() => onChange(on ? (value || []).filter(x => x !== c.value) : f.choices.map(x => x.value).filter(x => x === c.value || (value || []).includes(x)))}>{c.label}</button>; })}</span>;
    else if (f.type === "sel") ctl = <select className="tip-sel" value={String(value)} onChange={e => { const c = f.choices.find(c => String(c.value) === e.target.value); onChange(c ? c.value : e.target.value); }}>{f.choices.map(c => <option key={String(c.value)} value={String(c.value)}>{c.label}</option>)}</select>;
    else ctl = <span className="tip-num"><input type="number" inputMode="decimal" min={f.min} max={f.max} step={f.step || 1} value={value} style={{ width: (Math.max(2, String(value).length) + 0.6) + "ch" }} onChange={e => onChange(e.target.value)} />{f.unit ? <em>{f.unit.trim()}</em> : null}</span>;
    return <label className="tip-tile edit"><span>{f.label}</span>{ctl}</label>;
  }

  function Bars({ values }) {
    const vals = values.slice(-30);
    if (!vals.length) return <div className="tip-empty">Play a session to see your progress</div>;
    const W = 300, H = 96, n = vals.length, gap = 3;
    const lo = Math.min(0, ...vals), hi = Math.max(0, ...vals), span = hi - lo || 1;
    const zero = 6 + (hi / span) * (H - 12);
    const bw = Math.min(16, Math.max(2, (W - gap * (n - 1)) / Math.max(n, 6)));
    return (
      <svg className="tip-bars" viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" role="img" aria-label="Session results">
        <line x1="0" x2={W} y1={zero} y2={zero} className="tip-zero" />
        {vals.map((v, i) => {
          const h = Math.max(1.5, Math.abs(v) / span * (H - 12));
          const y = v >= 0 ? zero - h : zero;
          return <rect key={i} x={i * (bw + gap)} y={y} width={bw} height={h} rx="2" className={v >= 0 ? "up" : "dn"} style={{ animationDelay: (i * 0.025) + "s", transformOrigin: `0 ${zero}px` }} />;
        })}
      </svg>
    );
  }


  /* ---------- line chart with hover, styled like the Mental Math Trainer stats ---------- */
  function niceTicks(lo, hi, n, int) {
    if (!isFinite(lo) || !isFinite(hi)) return [0, 1];
    if (lo === hi) { lo -= 1; hi += 1; }
    const raw = (hi - lo) / Math.max(1, n), mag = 10 ** Math.floor(Math.log10(raw)), f = raw / mag;
    let step = (f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10) * mag;
    if (int) step = Math.max(1, Math.round(step));
    const a = Math.floor(lo / step) * step, b = Math.ceil(hi / step) * step, out = [];
    for (let v = a; v <= b + step * 1e-6; v += step) out.push(Math.round(v / step) * step);
    return out;
  }
  function monotone(pts) {   /* monotone cubic path through points, like d3 curveMonotoneX */
    const n = pts.length;
    if (n < 2) return n ? `M${pts[0][0]},${pts[0][1]}` : "";
    if (n === 2) return `M${pts[0][0]},${pts[0][1]}L${pts[1][0]},${pts[1][1]}`;
    const dx = [], dy = [], m = [];
    for (let i = 0; i < n - 1; i++) { dx.push(pts[i + 1][0] - pts[i][0]); dy.push(pts[i + 1][1] - pts[i][1]); m.push(dy[i] / dx[i]); }
    const t = [m[0]];
    for (let i = 1; i < n - 1; i++) t.push(m[i - 1] * m[i] <= 0 ? 0 : (3 * (dx[i - 1] + dx[i])) / ((2 * dx[i] + dx[i - 1]) / m[i - 1] + (dx[i] + 2 * dx[i - 1]) / m[i]));
    t.push(m[n - 2]);
    let d = `M${pts[0][0]},${pts[0][1]}`;
    for (let i = 0; i < n - 1; i++) {
      const h = dx[i] / 3;
      d += `C${pts[i][0] + h},${pts[i][1] + h * t[i]} ${pts[i + 1][0] - h},${pts[i + 1][1] - h * t[i + 1]} ${pts[i + 1][0]},${pts[i + 1][1]}`;
    }
    return d;
  }
  /* one colour per difficulty when a chart splits its sessions into lines */
  const GCOL = { Easy: "var(--tip-c3)", Medium: "var(--tip-c1)", Hard: "var(--tip-c5)", Custom: "var(--tip-c4)" }, GEXTRA = ["var(--tip-c2)", "var(--tip-c6)"];
  function LineChart({ title, sub, data, fmt, label, height, zero, avgLabel }) {
    /* data: [{ ts, v, note, g }] oldest first. With two or more groups (g, the difficulty) each group gets its own line,
       and the legend under the chart turns lines off and on; hidden lines leave the axes. */
    const box = useRef(null);
    const [w, setW] = useState(460);
    const [hov, setHov] = useState(null);
    const [off, setOff] = useState([]);
    const groups = useMemo(() => { const seen = []; data.forEach(d => { if (d.g != null && typeof d.v === "number" && isFinite(d.v) && !seen.includes(d.g)) seen.push(d.g); });
      const order = Object.keys(GCOL); return seen.sort((a, b) => (order.indexOf(a) < 0 ? 99 : order.indexOf(a)) - (order.indexOf(b) < 0 ? 99 : order.indexOf(b)) || String(a).localeCompare(String(b))); }, [data]);
    const multi = groups.length > 1;
    const colOf = g => GCOL[g] || GEXTRA[Math.max(0, groups.filter(x => !GCOL[x]).indexOf(g)) % GEXTRA.length];
    const toggle = g => { setHov(null); setOff(o => o.includes(g) ? o.filter(x => x !== g) : [...o, g]); };
    useEffect(() => {
      if (!box.current || !window.ResizeObserver) return;
      const ro = new ResizeObserver(es => { const cw = es[0].contentRect.width; if (cw > 0) setW(cw); });
      ro.observe(box.current);
      return () => ro.disconnect();
    }, []);
    const H = height || 190, ml = 44, mr = 14, mt = 10, mb = 26;
    const pts = data.filter(d => typeof d.v === "number" && isFinite(d.v) && !(multi && off.includes(d.g)));
    const roll = multi ? [] : pts.map((d, i) => { const a = pts.slice(Math.max(0, i - 4), i + 1); return a.reduce((x, y) => x + y.v, 0) / a.length; });
    const legendM = multi ? <div className="tip-legend tip-legend-g">{groups.map(g => <button type="button" key={g} className={"tip-lg" + (off.includes(g) ? " off" : "")} aria-pressed={!off.includes(g)}
      title={(off.includes(g) ? "Show " : "Hide ") + g} onClick={() => toggle(g)}><i style={{ background: colOf(g) }}></i>{g}</button>)}</div> : null;
    const f = fmt || ((v, d) => ie(v, d == null ? 0 : d));
    if (pts.length === 0) return (
      <div className="tip-cbox"><div className="tip-ch"><h4>{title}</h4>{sub ? <span>{sub}</span> : null}</div><div className="tip-empty">{multi ? "Every line is hidden. Tap a difficulty below to show it." : "Play a session to see your progress"}</div>{legendM}</div>
    );
    const all = pts.map(d => d.v).concat(roll);
    let lo = Math.min(...all), hi = Math.max(...all);
    if (zero) { lo = Math.min(0, lo); hi = Math.max(0, hi); }
    if (lo === hi) { const pad = Math.max(1, Math.abs(lo) * 0.2); hi += pad; lo = all.every(v => v >= 0) ? Math.max(0, lo - pad) : lo - pad; }
    else if (all.every(v => v >= 0)) lo = Math.max(0, lo);
    const ticks = niceTicks(lo, hi, 4, pts.every(d => Number.isInteger(d.v))), y0 = ticks[0], y1 = ticks[ticks.length - 1];
    const tstep = ticks.length > 1 ? ticks[1] - ticks[0] : 1, tdec = (String(+tstep.toFixed(6)).split(".")[1] || "").length;
    const oneDay = pts.length && new Date(pts[0].ts).toDateString() === new Date(pts[pts.length - 1].ts).toDateString();
    const xl = i => oneDay ? "#" + (i + 1) : fmtDate(pts[i].ts);
    const iw = Math.max(40, w - ml - mr), ih = H - mt - mb;
    const X = i => ml + (pts.length === 1 ? iw / 2 : (i / (pts.length - 1)) * iw);
    const Y = v => mt + ih - ((v - y0) / ((y1 - y0) || 1)) * ih;
    const line = multi ? null : monotone(pts.map((d, i) => [X(i), Y(d.v)]));
    const avg = !multi && pts.length > 2 ? monotone(roll.map((v, i) => [X(i), Y(v)])) : null;
    const series = multi ? groups.filter(g => !off.includes(g)).map(g => { const idx = pts.map((d, i) => d.g === g ? i : -1).filter(i => i >= 0);
      return { g, idx, d: idx.length > 1 ? monotone(idx.map(i => [X(i), Y(pts[i].v)])) : null }; }) : [];
    const every = Math.max(1, Math.ceil(pts.length / Math.max(1, Math.floor(iw / 64))));
    const onMove = e => {
      const r = e.currentTarget.getBoundingClientRect(), x = (e.clientX - r.left) * (w / r.width);
      let best = 0, bd = 1e9; pts.forEach((d, i) => { const dd = Math.abs(X(i) - x); if (dd < bd) { bd = dd; best = i; } });
      setHov(best);
    };
    const hp = hov != null ? pts[hov] : null;
    const TW = 190, tipLeft = hov != null ? Math.max(0, Math.min(X(hov) + 14 + TW > w ? X(hov) - 14 - TW : X(hov) + 14, w - TW)) : 0;
    return (
      <div className="tip-cbox">
        <div className="tip-ch"><h4>{title}</h4>{sub ? <span>{sub}</span> : null}</div>
        <div className="tip-cwrap" ref={box} style={{ height: H }}>
          <svg width={w} height={H} viewBox={`0 0 ${w} ${H}`} onPointerMove={onMove} onPointerLeave={() => setHov(null)} onPointerDown={onMove}>
            {ticks.map(t => <g key={t}><line x1={ml} x2={w - mr} y1={Y(t)} y2={Y(t)} className={"tip-grid-l" + (t === 0 && zero ? " zero" : "")} /><text x={ml - 8} y={Y(t) + 4} textAnchor="end" className="tip-axis">{f(t, tdec)}</text></g>)}
            <line x1={ml} x2={w - mr} y1={mt + ih} y2={mt + ih} className="tip-axis-l" />
            {pts.map((d, i) => (i % every === 0 && (i + every <= pts.length - 1 || i === pts.length - 1)) || i === pts.length - 1 ? <text key={i} x={X(i)} y={H - 6} textAnchor={pts.length > 1 && i === pts.length - 1 ? "end" : pts.length > 1 && i === 0 ? "start" : "middle"} className="tip-axis">{xl(i)}</text> : null)}
            {hov != null ? <rect x={X(hov) - Math.max(6, iw / pts.length / 2)} y={mt} width={Math.max(12, iw / pts.length)} height={ih} className="tip-cursor" /> : null}
            {avg ? <path d={avg} className="tip-l2" pathLength="1" /> : null}
            {multi ? series.map(S => S.d ? <path key={"l" + S.g + off.length} d={S.d} className="tip-l1" pathLength="1" style={{ stroke: colOf(S.g) }} /> : null) : <path d={line} className="tip-l1" pathLength="1" />}
            {pts.map((d, i) => <circle key={i + "-" + off.length} cx={X(i)} cy={Y(d.v)} r={hov === i ? 5 : 3} className={"tip-dot1" + (hov === i ? " on" : "")}
              style={{ animationDelay: (0.3 + i * 0.02) + "s", ...(multi ? { stroke: colOf(d.g), fill: hov === i ? colOf(d.g) : undefined } : {}) }} />)}
          </svg>
          {hp ? (
            <div className="tip-tip" style={{ left: tipLeft, top: 6, width: TW }}>
              <div className="tip-tip-h">{(oneDay ? "Session " + (hov + 1) + " · " : "") + fmtWhen(hp.ts) + (hp.note ? " · " + hp.note : "")}</div>
              <div className="tip-tip-r"><i className="c1" style={multi ? { background: colOf(hp.g) } : undefined}></i>{multi ? hp.g + " · " : ""}{label || "Value"}: <b>{f(hp.v)}</b></div>
              {avg ? <div className="tip-tip-r"><i className="c2"></i>{avgLabel || "5-session avg"}: <b>{f(roll[hov])}</b></div> : null}
            </div>
          ) : null}
        </div>
        {avg ? <div className="tip-legend"><span><i className="c1"></i>per session</span><span><i className="c2 dash"></i>{avgLabel || "5-session avg"}</span></div> : null}
        {legendM}
      </div>
    );
  }

  /* older Market Games: same settings / statistics screen as the Trading Game */
  function adaptGames(list, comps) {
    list.forEach((g, i) => {
      if (g.tip || !comps[g.id]) return;
      const fields = (g.fields || []).map(f => {
        const m = /^(.*?)\s*\(s\)$/.exec(f.label);
        return { ...f, type: "num", label: m ? m[1] : f.label, unit: m ? "s" : undefined };
      });
      const opts = (g.options || []).map(o => ({ key: o.key, label: o.label, type: "sel", choices: o.choices }));
      list[i] = { ...g, tip: true, fields: [...fields, ...opts], component: comps[g.id] };
    });
  }


  /* ---------- lessons: short reading with a worked example and a check question ---------- */
  function loadDone(id) { try { return JSON.parse(localStorage.getItem("tip-lessons-" + id) || "[]"); } catch (e) { return []; } }
  function saveDone(id, v) { try { localStorage.setItem("tip-lessons-" + id, JSON.stringify(v)); } catch (e) {} }
  const M = (text, as, cls) => TH(Rt, { text, as: as || "p", className: cls });
  function Check({ q, onRight }) {
    const [val, setVal] = useState("");
    const [res, setRes] = useState(null);
    const submit = v => {
      const given = v != null ? v : val;
      const ok = q.choices ? given === q.answer : Math.abs(TIP_DRILL.num(given) - q.answer) <= (q.tol != null ? q.tol : 1e-6);
      setRes({ ok, given }); if (ok) onRight();
    };
    return (
      <div className="tip-chk">
        <div className="tip-lbl">Check yourself</div>
        {M(q.q, "div", "tip-chk-q")}
        {q.choices ? (
          <div className="tip-chk-opts">{q.choices.map(c => <button key={c} type="button" className={"chip" + (res && res.given === c ? (res.ok ? " ok" : " bad") : "")} onClick={() => submit(c)}>{c}</button>)}</div>
        ) : (
          <div className="row nowrap"><input className="inp num" value={val} onChange={e => setVal(e.target.value)} onKeyDown={e => { if (e.key === "Enter") { e.preventDefault(); e.stopPropagation(); submit(); } }} placeholder="Answer" /><button className="ghost" onClick={() => submit()} disabled={!val.trim()}>Check</button></div>
        )}
        {res ? <div className={"box small " + (res.ok ? "okbox" : "errbox")}><b>{res.ok ? "Right. " : "Not quite. "}</b>{M(q.why, "span")}</div> : null}
      </div>
    );
  }
  function Lessons({ id, lessons }) {
    const [done, setDone] = useState(() => loadDone(id));
    const [open, setOpen] = useState(null);
    const mark = i => setDone(d => { if (d.includes(i)) return d; const n = [...d, i]; saveDone(id, n); return n; });
    const L = open != null ? lessons[open] : null;
    return (
      <section className="card tip-panel tip-lessons">
        <div className="tip-h-row"><h3 className="tip-h">Lessons</h3><span className="tip-cap">{done.length}/{lessons.length} done</span></div>
        <div className="tip-lesson-list">
          {lessons.map((l, i) => (
            <button key={i} type="button" className={"tip-lesson-btn" + (open === i ? " on" : "") + (done.includes(i) ? " done" : "")} onClick={() => setOpen(open === i ? null : i)}>
              <span className="n">{done.includes(i) ? "✓" : i + 1}</span><span className="t">{l.title}</span>
            </button>
          ))}
        </div>
        {L ? (
          <article className="tip-lesson" key={open}>
            <h4>{L.title}</h4>
            {L.body.map((p, k) => <div key={k}>{M(p)}</div>)}
            {L.formula ? <div className="tip-formula">{M("$$" + L.formula + "$$", "div")}</div> : null}
            {L.example ? <div className="tip-example"><div className="tip-lbl">Worked example</div>{L.example.map((p, k) => <div key={k}>{M(p)}</div>)}</div> : null}
            {L.check ? <Check key={"c" + open} q={L.check} onRight={() => mark(open)} /> : null}
            <div className="row">
              {!L.check && !done.includes(open) ? <button className="ghost sm" onClick={() => mark(open)}>Mark as done</button> : null}
              <span className="sp"></span>
              {open > 0 ? <button className="ghost sm" onClick={() => setOpen(open - 1)}>Previous</button> : null}
              {open < lessons.length - 1 ? <button className="start sm" onClick={() => setOpen(open + 1)}>Next lesson</button> : <button className="start sm" onClick={() => setOpen(null)}>Close</button>}
            </div>
          </article>
        ) : <p className="small muted tip-lesson-hint">Pick a lesson to read it. Each one ends with a quick check question; answer it right to tick the lesson off.</p>}
      </section>
    );
  }

  /* ---------- the shared trainer / game screen ---------- */
  function Setup({ game, go, back, backLabel }) {
    const { progress, recordGame, deletePlays } = Lt();
    /* a lesson's "Drill this" button can hand a config to the next opening of a trainer */
    const [pend] = useState(() => { const p = TIP.pending; if (p && p.id === game.id) { TIP.pending = null; return p.cfg; } return null; });
    const [diff, setDiff] = useState(pend ? "Custom" : "Easy");
    const [cfg, setCfg] = useState(() => ({ ...game.presets.Easy, ...(pend || {}) }));
    const clean = useMemo(() => clampCfg(game, cfg, diff), [game, cfg, diff]);
    const part = game.splitBy ? clean[game.splitBy] : null;
    const metric = game.metricOf ? game.metricOf(clean) : (game.metric || "score");
    const all = ((progress.games[game.id] && progress.games[game.id].plays) || []);
    const plays = all.filter(p => (!game.playFilter || game.playFilter(p)) && (!game.splitBy || (p[game.splitBy] || game.presets.Easy[game.splitBy]) === part));
    const valOf = p => metric === "pnl" ? p.pnl : p.score;
    const fmtVal = game.fmtVal ? (v => v == null ? "\u2014" : game.fmtVal(v, clean)) : v => v == null ? "—" : metric === "pnl" ? Be(v, 0) : ie(v, 0) + (Math.abs(v) === 1 && (game.unit || " pts") === " pts" ? " pt" : (game.unit || " pts"));
    const [playing, setPlaying] = useState(false);
    const [run, setRun] = useState(0);
    const [ask, setAsk] = useState(null);   /* {all:true} | {play} */
    const [showAll, setShowAll] = useState(false);
    const vals = plays.map(valOf).filter(v => typeof v === "number");
    const diffAt = useMemo(() => Object.fromEntries(plays.map(p => [p.ts, p.diff || "Other"])), [plays]);
    const best = vals.length ? Math.max(...vals) : null;
    const latest = vals.length ? vals[vals.length - 1] : null;
    const top = plays.filter(p => typeof valOf(p) === "number").slice().sort((a, b) => valOf(b) - valOf(a)).slice(0, 5);
    const choose = d => {
      setDiff(d);
      if (d !== "Custom") setCfg(c => { const n = { ...game.presets[d] }; game.fields.forEach(f => { if (f.always) n[f.key] = c[f.key]; }); return n; });
    };
    const [cd, setCd] = useState(0);        /* 3-2-1 countdown before every run, as in the Mental Math Trainer */
    const [cdGone, setCdGone] = useState(true);
    const err = game.check ? game.check(clean) : null;
    /* game.gate(progress) -> { locked, resume, review, note }: once-a-day games (Daily Challenge) */
    const ctxAll = Lt();
    const gate = game.gate ? game.gate(progress) : null;
    useEffect(() => { if (game.onOpen) game.onOpen(ctxAll); }, []);
    const start = () => {
      if (err) return;
      if (gate && gate.locked) { if (gate.review) { setRun(k => k + 1); setCd(0); setCdGone(true); setPlaying(true); window.scrollTo({ top: 0 }); } return; }
      const now = gate && gate.resume;
      setRun(k => k + 1); setCd(now ? 0 : 3); setCdGone(!!now); setPlaying(true); window.scrollTo({ top: 0 });
    };

  /* ---------- C++ in the browser: Clang (WebAssembly) in a worker, files in clang/ ----------
     The compiler (about 27 MB compressed) downloads once per visit and stays loaded between runs. */
  const cpp = (() => {
    let w = null, busy = null, ready = false;
    const mk = () => { w = new Worker(new URL("clang/worker.js", document.baseURI), { type: "module" }); w.onmessage = e => busy && busy(e.data); w.onerror = e => busy && busy({ t: "done", fatal: "The C++ compiler could not start" + (e && e.message ? ": " + e.message : ".") }); };
    const mb = n => (n / 1048576).toFixed(1);
    function warm() { if (w || typeof Worker === "undefined") return; mk(); busy = m => { if (m.t === "ready") { ready = true; busy = null; } }; w.postMessage({ warm: true }); }
    /* status(text) while it works; done(result) at the end */
    function run(code, status, done) {
      if (!w) mk();
      let timer = null, stage = ready ? "Compiling\u2026" : "Loading the C++ compiler\u2026";
      status(stage);
      const finish = r => { clearTimeout(timer); busy = null; ready = true; done(r); };
      busy = m => {
        if (m.t === "fetch") status("Downloading the C++ compiler (" + mb(m.done) + " / " + mb(m.total) + " MB, first run only)\u2026");
        else if (m.t === "ready") ready = true;
        else if (m.t === "compile") status(ready ? "Compiling\u2026" : "Loading the C++ compiler\u2026");
        else if (m.t === "run") { status("Running\u2026"); timer = setTimeout(() => { w.terminate(); w = null; ready = false; finish({ timeout: true, compileMs: m.compileMs }); }, 15000); }
        else if (m.t === "done") finish(m);
      };
      w.postMessage({ code });
    }
    return { run, warm };
  })();
