/* Sweet Spot: divided attention under time pressure, like the multi-target timing games of Arctic Shores-style
   assessments. Dials appear around the screen and expire; tap each one while its needle is inside the green arc. */
(function () {
  const { TH, TF, R } = TIP;
  const { useState, useEffect, useMemo, useRef } = R;
  const K = TIP_KIT;
  function Game({ settings: s, onFinish, onRestart }) {
    const [rng] = useState(() => jt());
    const g = useRef({ t: 0, dials: [], next: 0.3, hits: 0, early: 0, expired: 0, id: 0, fx: [] });
    const [, tick] = useState(0), [done, setDone] = useState(false);
    useEffect(() => {
      let raf, last = performance.now();
      const step = now => {
        const dt = Math.min(0.05, (now - last) / 1000); last = now; const G = g.current; G.t += dt;
        if (G.t >= s.secs) { setDone(true); return; }
        G.next -= dt;
        if (G.next <= 0 && G.dials.length < s.max) {
          let x, y; for (let t = 0; t < 30; t++) { x = rng.int(12, 88); y = rng.int(14, 86); if (G.dials.every(d => Math.hypot(d.x - x, (d.y - y) * 0.7) > 22)) break; }
          G.dials.push({ id: ++G.id, x, y, a: rng() * 360, sp: s.speed * (0.7 + rng() * 0.6) * (rng() < 0.5 ? -1 : 1), arc: rng() * 360, w: s.arc, life: s.life, age: 0 });
          G.next = s.every * (0.6 + rng() * 0.8);
        }
        G.dials.forEach(d => { d.a = (d.a + d.sp * dt + 360) % 360; d.age += dt; });
        G.dials = G.dials.filter(d => { if (d.age >= d.life) { G.expired++; G.fx.push({ x: d.x, y: d.y, t: G.t, k: "bad" }); return false; } return true; });
        G.fx = G.fx.filter(f => G.t - f.t < 0.5);
        tick(x => x + 1); raf = requestAnimationFrame(step);
      };
      raf = requestAnimationFrame(step); return () => cancelAnimationFrame(raf);
    }, []);
    function tap(id) {
      const G = g.current, d = G.dials.find(x => x.id === id); if (!d) return;
      const diff = ((d.a - d.arc) % 360 + 540) % 360 - 180;
      if (Math.abs(diff) <= d.w / 2) { G.hits++; G.fx.push({ x: d.x, y: d.y, t: G.t, k: "ok" }); G.dials = G.dials.filter(x => x.id !== id); }
      else { G.early++; G.fx.push({ x: d.x, y: d.y, t: G.t, k: "bad" }); d.age += d.life * 0.25; }
    }
    const G = g.current;
    const res = useMemo(() => done ? { score: G.hits * 2 - G.early - G.expired, hits: G.hits, early: G.early, expired: G.expired, correct: G.hits, total: G.hits + G.expired } : null, [done]);
    Jt(done, res, onFinish);
    if (done && res) return <K.Result headline={`${res.score} points`} sub={`${res.hits} hits · ${res.early} mistimed taps · ${res.expired} expired`} ring={res.total ? res.hits / res.total : 0} onRestart={onRestart}
      metrics={[{ label: "Hits", value: res.hits, tone: "up" }, { label: "Mistimed", value: res.early, tone: res.early ? "dn" : "" }, { label: "Expired", value: res.expired, tone: res.expired ? "dn" : "" }]}
      notes="2 per hit, −1 per mistimed tap or expired dial. Triage: serve the dial about to expire first, and pre-position your finger for the needle that will reach its arc next." />;
    return (
      <K.Stage title="Sweet Spot" sub="Tap a dial when its needle is in the green arc" clock={Math.max(0, s.secs - G.t)} clockFrac={1 - G.t / s.secs} stats={[{ label: "Hits", value: G.hits, pop: true, tone: "up" }, { label: "Missed", value: G.early + G.expired, tone: "dn" }]}>
        <div className="sw-field">
          {G.dials.map(d => {
            const life = 1 - d.age / d.life, a0 = (d.arc - d.w / 2 - 90) * Math.PI / 180, a1 = (d.arc + d.w / 2 - 90) * Math.PI / 180, na = (d.a - 90) * Math.PI / 180, C = 30, Rr = 23;
            return <button key={d.id} className="sw-dial" style={{ left: d.x + "%", top: d.y + "%" }} onPointerDown={e => { e.preventDefault(); tap(d.id); }}>
              <svg viewBox="0 0 60 60"><circle cx={C} cy={C} r="27" className="sw-face" /><circle cx={C} cy={C} r="28.5" className="sw-life" strokeDasharray={2 * Math.PI * 28.5} strokeDashoffset={2 * Math.PI * 28.5 * (1 - life)} transform={`rotate(-90 ${C} ${C})`} />
                <path d={`M${C + Rr * Math.cos(a0)},${C + Rr * Math.sin(a0)}A${Rr},${Rr} 0 0,1 ${C + Rr * Math.cos(a1)},${C + Rr * Math.sin(a1)}`} className="sw-arc" />
                <path d={`M${C},${C}L${C + 22 * Math.cos(na)},${C + 22 * Math.sin(na)}`} className="sw-needle" /><circle cx={C} cy={C} r="3" className="sw-hub" /></svg></button>;
          })}
          {G.fx.map((f, i) => <i key={i} className={"sw-fx " + f.k} style={{ left: f.x + "%", top: f.y + "%", opacity: 1 - (G.t - f.t) / 0.5 }} />)}
        </div>
      </K.Stage>
    );
  }
  TIP_TRAINERS.register({
    id: "sweet", name: "Sweet Spot", metric: "score", unit: " pts", version: 2, playFilter: K.versioned(2),
    blurb: "Divided attention under time pressure, like the multi-target timing games in Arctic Shores-style assessments: dials pop up around the screen and expire, and each scores only if you tap while its needle is inside the green arc.",
    fields: [
      { key: "secs", label: "Duration", type: "num", min: 20, max: 180, unit: "s" },
      { key: "max", label: "Dials at once", type: "num", min: 1, max: 6 },
      { key: "life", label: "Dial lifetime", type: "num", min: 2, max: 12, unit: "s" },
      { key: "arc", label: "Arc width", type: "num", min: 15, max: 90, unit: "°" },
      { key: "speed", label: "Needle speed", type: "num", min: 60, max: 400, unit: "°/s" },
    ],
    presets: { Easy: { secs: 40, max: 2, life: 7, arc: 60, speed: 140, every: 1.8 }, Medium: { secs: 60, max: 3, life: 5.5, arc: 45, speed: 190, every: 1.3 }, Hard: { secs: 75, max: 5, life: 4.5, arc: 34, speed: 240, every: 0.9 } },
    fix: c => { if (c.every == null) c.every = 1.3; },
    note: "A mistimed tap ages the dial by a quarter of its life.",
    component: Game,
  });
})();
