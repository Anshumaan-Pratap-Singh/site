    const n = seq.length, d = seq.slice(1).map((v, j) => v - seq[j]);
    if (d.every(x => x === d[0]) && seq[n - 1] + d[0] !== next) return true;
    if (seq.every(v => v !== 0)) { const q = seq.slice(1).map((v, j) => v / seq[j]); if (q.every(x => Math.abs(x - q[0]) < 1e-12) && Math.abs(seq[n - 1] * q[0] - next) > 1e-9) return true; }
    if (n >= 4) { const d2 = d.slice(1).map((v, j) => v - d[j]); if (d2.every(x => x === d2[0]) && seq[n - 1] + d[d.length - 1] + d2[0] !== next) return true; }
    return new Set(seq).size < 3;
  }
