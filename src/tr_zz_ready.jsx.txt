      else if (typeof p.total === "number" && p.total > 0 && typeof p.correct === "number") q = accQ(p);
      else if (typeof p.pnl === "number") q = MMQ(p);
      else if (typeof p.score === "number" && best > 0) q = clamp(p.score / best);

  const pad = n => String(n).padStart(2, "0");
  const dkey = ts => { const d = new Date(ts); return d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate()); };
  const seedOf = k => { let h = 2166136261; for (const c of "daily" + k) { h ^= c.charCodeAt(0); h = Math.imul(h, 16777619); } return (h >>> 0) % 2147483647; };
  function dgen(r, s, i, hist) {
    const part = PLAN[i % PLAN.length], D = TIP_TRAINERS.DEFS, OA = window.__TIP_OA;
    let q;
    if (part === "arith") q = D.frac.spec.gen(r, D.frac.presets.Medium, i, hist);
    else if (part === "seq") q = D.seq.spec.gen(r, D.seq.presets.Medium, i, hist);
    else if (part === "prob") q = OA.genBto(r, {}, i, hist);
    else q = OA.genObQ(r);
    return { ...q, kind: part + ":" + (q.kind == null ? "" : q.kind), part };
  }
  function streakOf(plays) {
    const days = new Set(plays.map(p => dkey(p.ts)));
    let cur = 0, d = new Date(); if (!days.has(dkey(d))) d.setDate(d.getDate() - 1);
    while (days.has(dkey(d))) { cur++; d.setDate(d.getDate() - 1); }
    const sorted = [...days].sort(); let best = 0, run = 0, prev = null;
    sorted.forEach(k => { const [y, m, dd] = k.split("-").map(Number), t = new Date(y, m - 1, dd).getTime(); run = prev != null && Math.round((t - prev) / DAY) === 1 ? run + 1 : 1; best = Math.max(best, run); prev = t; });
    return { cur, best, today: days.has(dkey(Date.now())), days: days.size };
  }
  /* one run a day: the run in progress is kept in progress.dailyRun = { day, t0, hist, done } (synced with the
     rest of progress), so reopening resumes at the first unanswered question; the clock keeps running while closed.
     A run left unfinished on an earlier day is scored with what was answered and dated the day it started. */
  const RUN = "dailyRun", SECS = 300;
  const resOf = hist => {
    const n = hist.length, c = hist.filter(h => h.ok).length, wrong = hist.filter(h => !h.ok && !h.skipped && !h.timedOut).length;
    const okMs = hist.filter(h => h.ok).map(h => h.ms); let best = 0, run = 0; hist.forEach(h => { run = h.ok ? run + 1 : 0; best = Math.max(best, run); });
    const parts = {}; Object.keys(PART).forEach(k => { const L = hist.filter(h => h.part === k); parts[k] = [L.filter(h => h.ok).length, L.length]; });
    return { score: c, correct: c, wrong, total: n, acc: n ? c / n : 0, avgMs: okMs.length ? okMs.reduce((a, b) => a + b, 0) / okMs.length : null, bestStreak: best, parts };

  const playedOn = (G, day) => ((G.games.daily && G.games.daily.plays) || []).some(p => dkey(p.ts) === day);
  function settleStale(G) {
    const r = G[RUN], today = dkey(Date.now());
    if (!r) return false;
    if (r.day === today) {
      if (r.done || Date.now() - r.t0 < SECS * 1000) return false;
      if (!playedOn(G, today)) { const D = G.games.daily || (G.games.daily = { plays: [] }); D.plays.push({ ts: r.t0, diff: "Easy", v: 1, ...resOf(r.hist || []) }); }
      r.done = true; return true;
    }
    if (!r.done && (r.hist || []).length && !playedOn(G, r.day)) {
      const D = G.games.daily || (G.games.daily = { plays: [] });
      D.plays.push({ ts: r.t0, diff: "Easy", v: 1, ...resOf(r.hist) }); D.plays.sort((a, b) => a.ts - b.ts);
    }
    delete G[RUN]; return true;
  }
  function dailyGate(progress) {

  const TAGS = [["slip", "Arithmetic slip"], ["misread", "Misread the question"], ["concept", "Concept gap"], ["time", "Ran out of time"]];

  const digits = v => String(Math.abs(v)).replace(/[^0-9]/g, "").split("").sort().join("");
  function guess(h) {
    if (h.timedOut || h.skipped) return "time";
    const g = TIP_DRILL.num(typeof h.given === "object" && h.given ? NaN : h.given), a = TIP_DRILL.num(h.answer);
    if (isFinite(g) && isFinite(a)) {
      const rel = Math.abs(g - a) / Math.max(Math.abs(a), 1e-9), ratio = a ? g / a : 0;
      if (rel <= 0.05 || [10, 100, 0.1, 0.01].some(x => Math.abs(ratio - x) < 1e-6) || Math.abs(g + a) < 1e-9 || (digits(g) === digits(a) && digits(a).length > 1)) return "slip";
      if (a > 0 && a < 1 && Math.abs(g - (1 - a)) < 0.01) return "misread";
    }
    return "concept";
  }
